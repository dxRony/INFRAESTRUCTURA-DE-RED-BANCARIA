########## 6) /opt/failover.sh (script de failover del anillo - v5 legacy con eth3,
########## NO ejecutandose actualmente; el nodo solo tiene eth0/eth1/eth2)

#!/bin/sh
# failover anillo v5 por reachability del ENLACE DIRECTO de cada vecino.
# Al caer el vecino se borran TODAS las rutas con nexthop hacia el (metric10 y metric20),
# para evitar ECMP/parpadeo hacia el enlace muerto. El vecino directo se desvia con /32
# por la otra punta. En up se restauran primarias (m10) y las metric20 simetricas.
s3=up
s2=up

clean_eth3() {
  for s in 10.0.0.4/30 10.0.0.8/30 10.0.0.12/30; do ip route del $s via 10.0.0.2 dev eth3 2>/dev/null; done
  ip route del 10.0.0.12/30 via 10.0.0.2 dev eth3 metric 20 2>/dev/null
  ip route del 10.0.0.16/30 via 10.0.0.2 dev eth3 metric 20 2>/dev/null
}
restore_eth3() {
  for s in 10.0.0.4/30 10.0.0.8/30 10.0.0.12/30; do ip route add $s via 10.0.0.2 dev eth3 metric 10 2>/dev/null; done
  ip route add 10.0.0.12/30 via 10.0.0.2 dev eth3 metric 20 2>/dev/null
  ip route add 10.0.0.16/30 via 10.0.0.2 dev eth3 metric 20 2>/dev/null
}
clean_eth2() {
  for s in 10.0.0.0/30 10.0.0.4/30 10.0.0.8/30 10.0.0.12/30; do ip route del $s via 10.0.0.17 dev eth2 2>/dev/null; done
}
restore_eth2() {
  for s in 10.0.0.0/30 10.0.0.4/30 10.0.0.8/30 10.0.0.12/30; do ip route add $s via 10.0.0.17 dev eth2 metric 20 2>/dev/null; done
}

while :; do
  # --- lado eth3 / Banco 2 (10.0.0.2) ---
  if [ "$s3" = "up" ]; then
    if ping -c1 -W2 -I eth3 10.0.0.2 >/dev/null 2>&1; then
      restore_eth3
    else
      clean_eth3
      ip route replace 10.0.0.2/32 via 10.0.0.17 dev eth2 metric 5 2>/dev/null
      s3=down
    fi
  else
    ip route del 10.0.0.2/32 2>/dev/null
    sleep 1
    if ping -c1 -W2 -I eth3 10.0.0.2 >/dev/null 2>&1; then
      restore_eth3
      s3=up
    else
      clean_eth3
      ip route replace 10.0.0.2/32 via 10.0.0.17 dev eth2 metric 5 2>/dev/null
    fi
  fi
  # --- lado eth2 / Banco 5 (10.0.0.17) ---
  if [ "$s2" = "up" ]; then
    if ping -c1 -W2 -I eth2 10.0.0.17 >/dev/null 2>&1; then
      restore_eth2
    else
      clean_eth2
      ip route replace 10.0.0.17/32 via 10.0.0.2 dev eth3 metric 5 2>/dev/null
      s2=down
    fi
  else
    ip route del 10.0.0.17/32 2>/dev/null
    sleep 1
    if ping -c1 -W2 -I eth2 10.0.0.17 >/dev/null 2>&1; then
      restore_eth2
      s2=up
    else
      clean_eth2
      ip route replace 10.0.0.17/32 via 10.0.0.2 dev eth3 metric 5 2>/dev/null
    fi
  fi
  sleep 2
done
