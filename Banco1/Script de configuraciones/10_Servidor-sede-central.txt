#!/usr/bin/env python3
"""
Servidor DB - Banco 1 (con empleados, roles y auditoria)

Solo usa librerias estandar: http.server, sqlite3, json

import json
from datetime import datetime

from urllib.parse import urlparse, parse_qs

def get_conn():
    conn = sqlite3.connect(DB_FILE)
    conn.row_factory = sqlite3.Row
    return conn


def init_db():

    cur = conn.cursor()

        CREATE TABLE IF NOT EXISTS empleados (
            id INTEGER PRIMARY KEY AUTOINCREMENT,

            contrasena TEXT NOT NULL,
            nombre TEXT NOT NULL,
            rol TEXT NOT NULL CHECK (rol IN ('admin', 'cajero'))
        )
    """)

    cur.execute("""
        CREATE TABLE IF NOT EXISTS cuentas (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            nombre TEXT NOT NULL,
            saldo REAL NOT NULL,
            creado_por INTEGER,

            FOREIGN KEY (creado_por) REFERENCES empleados(id)
        )

            empleado_id INTEGER NOT NULL,

def get_empleado_por_usuario(usuario):
    conn = get_conn()
    row = conn.execute(
        "SELECT * FROM empleados WHERE usuario = ?", (usuario,)
    ).fetchone()
    conn.close()
    return dict(row) if row else None


def get_empleado_por_id(empleado_id):
    conn = get_conn()
    row = conn.execute(
        "SELECT * FROM empleados WHERE id = ?", (empleado_id,)
    ).fetchone()
    conn.close()
    return dict(row) if row else None


def es_admin(empleado_id):
    emp = get_empleado_por_id(empleado_id)
    return emp is not None and emp["rol"] == "admin"


def es_cajero(empleado_id):
    emp = get_empleado_por_id(empleado_id)
    return emp is not None and emp["rol"] == "cajero"


def get_cuenta(cuenta_id):
    conn = get_conn()
    row = conn.execute(
        "SELECT * FROM cuentas WHERE id = ?", (cuenta_id,)
    ).fetchone()
    conn.close()
    return dict(row) if row else None


def listar_cuentas():
    conn = get_conn()
    rows = conn.execute("""
        SELECT c.id, c.nombre, c.saldo
        FROM cuentas c
        ORDER BY c.id
    """).fetchall()
    conn.close()
    return [dict(r) for r in rows]


def crear_cuenta(nombre, saldo_inicial, empleado_id):
    conn = get_conn()
    ahora = datetime.now().isoformat(timespec="seconds")
    cur = conn.execute(
        "INSERT INTO cuentas (nombre, saldo) VALUES (?, ?)",
        (nombre, saldo_inicial, empleado_id, ahora),
    )
    conn.commit()
    nuevo_id = cur.lastrowid
    conn.close()
    return nuevo_id


def actualizar_saldo(cuenta_id, nuevo_saldo):
    conn = get_conn()
    conn.execute("UPDATE cuentas SET saldo = ? WHERE id = ?", (nuevo_saldo, cuenta_id))
    conn.commit()
    conn.close()


def registrar_transaccion(cuenta_id, tipo, monto, empleado_id, cuenta_destino=None):
    conn = get_conn()
    ahora = datetime.now().isoformat(timespec="seconds")
    conn.execute(
        "INSERT INTO transacciones (cuenta_id, cuenta_destino_id, tipo, monto, usuario_id, sucursal_id, fecha) "
        "VALUES (?, ?, ?, ?, ?, 1, ?)",
        (cuenta_id, cuenta_destino, tipo, monto, empleado_id, ahora),
    )
    conn.commit()
    conn.close()


def listar_transacciones():
    conn = get_conn()
    rows = conn.execute("""
        SELECT t.id, t.tipo, t.monto, t.fecha,
               c1.nombre AS cuenta_nombre,
               c2.nombre AS cuenta_destino_nombre,
               e.nombre AS empleado_nombre, e.usuario AS empleado_usuario
        FROM transacciones t
        JOIN cuentas c1 ON t.cuenta_id = c1.id
        LEFT JOIN cuentas c2 ON t.cuenta_destino_id = c2.id
        LEFT JOIN empleados e ON t.usuario_id = e.id
        ORDER BY t.id DESC
    """).fetchall()
    conn.close()
    return [dict(r) for r in rows]


def crear_empleado(usuario, contrasena, nombre, rol):
    conn = get_conn()
    try:
        cur = conn.execute(
            "INSERT INTO empleados (usuario, contrasena, nombre, rol) VALUES (?, ?, ?, ?)",
            (usuario, contrasena, nombre, rol),
        )
        conn.commit()
        nuevo_id = cur.lastrowid
        conn.close()
        return nuevo_id, None
    except sqlite3.IntegrityError:
        conn.close()
        return None, "el usuario ya existe"


def listar_empleados():
    conn = get_conn()
    rows = conn.execute(
        "SELECT id, usuario, nombre, rol FROM empleados ORDER BY id"
    ).fetchall()
    conn.close()
    return [dict(r) for r in rows]


class DBHandler(BaseHTTPRequestHandler):

    def _send_json(self, status, data):
        body = json.dumps(data).encode("utf-8")
        self.send_response(status)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def _read_json_body(self):
        length = int(self.headers.get("Content-Length", 0))
        raw = self.rfile.read(length)
        return json.loads(raw) if raw else {}

    def do_GET(self):
        parsed = urlparse(self.path)
        params = parse_qs(parsed.query)

        if parsed.path == "/saldo":
            if "id" not in params:
                self._send_json(400, {"error": "falta parametro id"})
                return
            cuenta = get_cuenta(int(params["id"][0]))
            if cuenta is None:
                self._send_json(404, {"error": "cuenta no existe"})
            else:
                self._send_json(200, cuenta)

        elif parsed.path == "/cuentas":
            self._send_json(200, listar_cuentas())

        elif parsed.path == "/empleados":
            empleado_id = params.get("empleado_id", [None])[0]
            if not empleado_id or not es_admin(int(empleado_id)):
                self._send_json(403, {"error": "solo un admin puede ver empleados"})
                return
            self._send_json(200, listar_empleados())

        elif parsed.path == "/transacciones":
            empleado_id = params.get("empleado_id", [None])[0]
            if not empleado_id or not es_admin(int(empleado_id)):
                self._send_json(403, {"error": "solo un admin puede ver el reporte"})
                return
            self._send_json(200, listar_transacciones())

        else:
            self._send_json(404, {"error": "ruta no encontrada"})

    def do_POST(self):
        parsed = urlparse(self.path)
        try:
            data = self._read_json_body()
        except json.JSONDecodeError:
            self._send_json(400, {"error": "JSON invalido"})
            return

        if parsed.path == "/login":
            usuario = data.get("usuario", "")
            contrasena = data.get("contrasena", "")
            emp = get_empleado_por_usuario(usuario)
            if emp is None or emp["contrasena"] != contrasena:
                self._send_json(401, {"error": "usuario o contrasena incorrectos"})
                return
            self._send_json(200, {
                "id": emp["id"], "nombre": emp["nombre"], "rol": emp["rol"],
            })

        elif parsed.path == "/clientes":
            empleado_id = data.get("empleado_id")
            nombre = data.get("nombre")
            saldo_inicial = data.get("saldo_inicial", 0)
            if not empleado_id or not es_admin(empleado_id):
                self._send_json(403, {"error": "solo un admin puede crear clientes"})
                return
            if not nombre:
                self._send_json(400, {"error": "falta el nombre del cliente"})
                return
            nuevo_id = crear_cuenta(nombre, float(saldo_inicial), empleado_id)
            self._send_json(200, {"id": nuevo_id, "nombre": nombre, "saldo": float(saldo_inicial)})

        elif parsed.path == "/empleados":
            empleado_id = data.get("empleado_id")
            if not empleado_id or not es_admin(empleado_id):
                self._send_json(403, {"error": "solo un admin puede crear usuarios"})
                return
            usuario = data.get("usuario")
            contrasena = data.get("contrasena")
            nombre = data.get("nombre")
            rol = data.get("rol")
            if not all([usuario, contrasena, nombre, rol]) or rol not in ("admin", "cajero"):
                self._send_json(400, {"error": "datos invalidos"})
                return
            nuevo_id, error = crear_empleado(usuario, contrasena, nombre, rol)
            if error:
                self._send_json(400, {"error": error})
            else:
                self._send_json(200, {"id": nuevo_id, "usuario": usuario, "rol": rol})

        elif parsed.path == "/deposito":
            self._operacion(data, "deposito")

        elif parsed.path == "/retiro":
            self._operacion(data, "retiro")

        elif parsed.path == "/transferencia":
            self._transferencia(data)

        else:
            self._send_json(404, {"error": "ruta no encontrada"})

    def _operacion(self, data, tipo):
        empleado_id = data.get("empleado_id")
        cuenta_id = data.get("id")
        monto = data.get("monto")

        if not empleado_id or not es_cajero(empleado_id):
            self._send_json(403, {"error": "solo un cajero puede hacer esta operacion"})
            return
        if cuenta_id is None or monto is None or monto <= 0:
            self._send_json(400, {"error": "datos invalidos"})
            return

        cuenta = get_cuenta(cuenta_id)
        if cuenta is None:
            self._send_json(404, {"error": "cuenta no existe"})
            return

        if tipo == "deposito":
            nuevo_saldo = cuenta["saldo"] + monto
        else:
            if cuenta["saldo"] < monto:
                self._send_json(400, {"error": "fondos insuficientes"})
                return
            nuevo_saldo = cuenta["saldo"] - monto

        actualizar_saldo(cuenta_id, nuevo_saldo)
        registrar_transaccion(cuenta_id, tipo, monto, empleado_id)
        self._send_json(200, {"id": cuenta_id, "saldo": nuevo_saldo})

    def _transferencia(self, data):
        empleado_id = data.get("empleado_id")
        origen_id = data.get("origen")
        destino_id = data.get("destino")
        monto = data.get("monto")

        if not empleado_id or not es_cajero(empleado_id):
            self._send_json(403, {"error": "solo un cajero puede hacer transferencias"})
            return
        if not all([origen_id, destino_id, monto]) or monto <= 0:
            self._send_json(400, {"error": "datos invalidos"})
            return
        if origen_id == destino_id:
            self._send_json(400, {"error": "origen y destino no pueden ser la misma cuenta"})
            return

        origen = get_cuenta(origen_id)
        destino = get_cuenta(destino_id)
        if origen is None or destino is None:
            self._send_json(404, {"error": "cuenta origen o destino no existe"})
            return
        if origen["saldo"] < monto:
            self._send_json(400, {"error": "fondos insuficientes"})
            return

        actualizar_saldo(origen_id, origen["saldo"] - monto)
        actualizar_saldo(destino_id, destino["saldo"] + monto)
        registrar_transaccion(origen_id, "transferencia", monto, empleado_id, cuenta_destino=destino_id)

        self._send_json(200, {
            "origen": {"id": origen_id, "saldo": origen["saldo"] - monto},
            "destino": {"id": destino_id, "saldo": destino["saldo"] + monto},
        })

    def log_message(self, format, *args):
        print("[DB] %s" % (format % args))


if __name__ == "__main__":
    init_db()
    server = HTTPServer(("0.0.0.0", PORT), DBHandler)
    print(f"Servidor DB escuchando en puerto {PORT}...")
    server.serve_forever()
